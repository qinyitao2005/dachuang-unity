using System;
using System.IO;
using System.Linq;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;

namespace RoboticsLab.Editor
{
    public static class LabRobotBuilder
    {
        const string Generated="Assets/RoboticsLab/Generated";
        const string OriginalScene="Assets/Scenes/SampleScene.unity";
        static readonly string[] Labels={"01  ·  轨迹绘制机械臂","02  ·  精密操作机械臂","03  ·  综合实训机械臂"};
        public static void ExtractOriginal()
        {
            Directory.CreateDirectory(Generated);AssetDatabase.Refresh();
            bool complete=true;for(int i=0;i<3;i++)complete &= AssetDatabase.LoadAssetAtPath<GameObject>(PrefabPath(i))!=null;
            if(complete)return;
            var scene=EditorSceneManager.OpenScene(OriginalScene,OpenSceneMode.Single);
            var original=UnityEngine.Object.FindObjectsOfType<IK>(true)
                .Where(x=>x.gameObject.scene==scene&&x.joints!=null&&x.joints.Length>=5&&x.endEffector)
                .OrderBy(x=>HierarchyPath(x.transform)).ToArray();
            if(original.Length<3)throw new Exception("Expected three configured mechanical arms in SampleScene; found "+original.Length);
            for(int i=0;i<3;i++)
            {
                var source=original[i];Transform root=source.transform;
                if(root==source.joints[0].joint&&root.parent)root=root.parent;
                var container=new GameObject("OriginalRobot"+(i+1));
                var clone=UnityEngine.Object.Instantiate(root.gameObject);clone.name="OriginalModel";clone.transform.SetParent(container.transform,true);
                foreach(var behaviour in clone.GetComponentsInChildren<MonoBehaviour>(true))
                    if(behaviour is IK || behaviour is IKnew || behaviour is IKnew2 || behaviour is AnimationEvents || behaviour is ArmPoseSaver || behaviour is CubeMove)behaviour.enabled=false;
                foreach(var animator in clone.GetComponentsInChildren<Animator>(true))animator.enabled=false;
                foreach(var collider in clone.GetComponentsInChildren<Collider>(true))UnityEngine.Object.DestroyImmediate(collider);
                PrefabUtility.SaveAsPrefabAsset(container,PrefabPath(i));
                Debug.Log("ROBOT_LAB_EXTRACT "+i+" source="+HierarchyPath(root)+" meshes="+clone.GetComponentsInChildren<MeshFilter>(true).Length);
                UnityEngine.Object.DestroyImmediate(container);
            }
            AssetDatabase.SaveAssets();
        }
        static string PrefabPath(int index)=>Generated+"/OriginalRobot"+(index+1)+".prefab";
        static string HierarchyPath(Transform t)=>t.parent?HierarchyPath(t.parent)+"/"+t.name:t.name;
        public static LabRobotStation[] Build()
        {
            var result=new LabRobotStation[3];
            for(int i=0;i<3;i++)
            {
                var go=GameObject.Find("Station0"+(i+1));
                if(!go){go=new GameObject("Station0"+(i+1));go.transform.position=new Vector3(-4.5f,0,5.5f+i*5.5f);}
                var station=go.GetComponent<LabRobotStation>()??go.AddComponent<LabRobotStation>();
                station.stationIndex=i;station.displayName=Labels[i];
                var prefab=AssetDatabase.LoadAssetAtPath<GameObject>(PrefabPath(i));
                if(!prefab)throw new Exception("Run ExtractOriginal before Build: "+PrefabPath(i));
                var model=(GameObject)PrefabUtility.InstantiatePrefab(prefab,go.transform);model.name="RobotModel";
                model.transform.localPosition=Vector3.zero;model.transform.localRotation=Quaternion.identity;
                Bounds bounds=GetBounds(model);float scale=1.28f/Mathf.Max(.1f,bounds.size.y);
                model.transform.localScale=Vector3.one*scale;
                bounds=GetBounds(model);
                model.transform.position+=go.transform.TransformPoint(new Vector3(-.25f,.93f,0))-new Vector3(bounds.center.x,bounds.min.y,bounds.center.z);
                var solver=model.GetComponentsInChildren<IK>(true).FirstOrDefault(x=>x.endEffector);
                if(!solver)throw new Exception("Robot prefab missing configured IK: "+i);
                station.modelRoot=model.transform;station.endEffector=solver.endEffector;
                station.joints=solver.joints.Where(x=>x!=null&&x.joint).Select((x,n)=>new JointConfig
                {
                    joint=x.joint,axis=x.axis,
                    // Original Euler limits assumed a zero pose. The adapter uses offsets from the preserved pose.
                    minAngle=n==0?-160:n==solver.joints.Length-1?-100:-85,
                    maxAngle=n==0?160:n==solver.joints.Length-1?100:85
                }).ToArray();
                // Orient the existing flange toward the accessible front edge (+X).
                if(station.joints.Length>0)
                {
                    Vector3 pivot=station.joints[0].joint.position;
                    Vector3 from=Vector3.ProjectOnPlane(station.endEffector.position-pivot,Vector3.up);
                    if(from.sqrMagnitude>.0001f)
                    {
                        float angle=Vector3.SignedAngle(from,Vector3.right,Vector3.up);
                        model.transform.RotateAround(pivot,Vector3.up,angle);
                    }
                }
                bounds=GetBounds(model);
                // Place the robot foot on the worktop again after normalization.
                model.transform.position+=Vector3.up*(go.transform.position.y+.93f-bounds.min.y);
                if(station.joints.Length>0)
                {
                    Vector3 mountingCenter=go.transform.TransformPoint(new Vector3(-.63f,0,0));
                    Vector3 foot=station.joints[0].joint.position;
                    model.transform.position+=new Vector3(mountingCenter.x-foot.x,0,mountingCenter.z-foot.z);
                }
                Vector3 tip=go.transform.InverseTransformPoint(station.endEffector.position);
                station.exerciseCenter=new Vector3(Mathf.Clamp(tip.x,.1f,.85f),Mathf.Clamp(tip.y,1.1f,1.55f),tip.z);
                station.drawingHeight=Mathf.Max(1.015f,station.exerciseCenter.y-.12f);
                station.workspaceHalfSize=new Vector3(.33f,.38f,.48f);
                station.toolTarget=new GameObject("ToolTarget").transform;station.toolTarget.SetParent(go.transform,false);station.toolTarget.localPosition=station.exerciseCenter;
                MakeExercises(station);
                foreach(var legacy in model.GetComponentsInChildren<MonoBehaviour>(true))
                    if(legacy is IK||legacy is IKnew||legacy is IKnew2||legacy is AnimationEvents||legacy is ArmPoseSaver||legacy is CubeMove)legacy.enabled=false;
                result[i]=station;
                Debug.Log("ROBOT_LAB_STATION "+i+" model="+bounds+" tip="+tip+" exercise="+station.exerciseCenter+" joints="+station.joints.Length);
            }
            return result;
        }
        static void MakeExercises(LabRobotStation s)
        {
            Color cyan=new Color(.06f,.57f,.69f),dark=new Color(.11f,.18f,.23f);
            Vector3 c=s.exerciseCenter;
            var board=new GameObject("DrawingExercise");board.transform.SetParent(s.transform,false);s.drawingBoard=board.transform;
            var surface=Cube("DrawingPaper",board.transform,new Vector3(c.x,s.drawingHeight-.009f,c.z),new Vector3(.7f,.016f,1.02f),Color.white);
            Cube("BoardBacking",board.transform,new Vector3(c.x,s.drawingHeight-.035f,c.z),new Vector3(.77f,.035f,1.09f),dark);
            // Four adjustable supports connect the elevated drawing board to the physical worktop.
            float supportTop=s.drawingHeight-.0525f;
            float supportBottom=.902f;
            float supportHeight=Mathf.Max(.035f,supportTop-supportBottom);
            for(int x=-1;x<=1;x+=2)for(int z=-1;z<=1;z+=2)
            {
                Vector3 foot=new Vector3(c.x+x*.29f,supportBottom,c.z+z*.43f);
                Cube("BoardSupportFoot",board.transform,foot+Vector3.up*.006f,new Vector3(.105f,.012f,.105f),dark);
                Cube("BoardSupportPost",board.transform,foot+Vector3.up*(supportHeight*.5f),new Vector3(.029f,supportHeight,.029f),new Color(.6f,.66f,.7f));
                Cube("BoardSupportSleeve",board.transform,foot+Vector3.up*(supportHeight*.23f),new Vector3(.041f,supportHeight*.44f,.041f),new Color(.22f,.29f,.34f));
            }
            for(int k=-4;k<=4;k++)
            {
                var a=new Vector3(c.x-.33f,s.drawingHeight+.001f,c.z+k*.1f);var b=a+Vector3.right*.66f;
                MakeLine("PaperGrid",board.transform,new[]{a,b},.0013f,new Color(.72f,.84f,.87f));
            }
            for(int k=-3;k<=3;k++)
            {
                var a=new Vector3(c.x+k*.1f,s.drawingHeight+.001f,c.z-.49f);var b=a+Vector3.forward*.98f;
                MakeLine("PaperGrid",board.transform,new[]{a,b},.0013f,new Color(.72f,.84f,.87f));
            }
            var pen=new GameObject("PenTool");pen.transform.SetParent(s.transform,false);s.penRoot=pen.transform;
            var nib=GameObject.CreatePrimitive(PrimitiveType.Cylinder);nib.name="PenNib";nib.transform.SetParent(pen.transform,false);nib.transform.localPosition=new Vector3(0,.028f,0);nib.transform.localScale=new Vector3(.014f,.028f,.014f);SetMaterial(nib,dark);UnityEngine.Object.DestroyImmediate(nib.GetComponent<Collider>());
            var wire=new GameObject("WireExercise");wire.transform.SetParent(s.transform,false);s.wireRoot=wire.transform;
            const int count=121;s.wirePoints=new Vector3[count];
            for(int k=0;k<count;k++)
            {
                float t=k/(float)(count-1);
                s.wirePoints[k]=new Vector3(c.x+.115f*Mathf.Sin(t*Mathf.PI*4),c.y+.13f*Mathf.Sin(t*Mathf.PI*2),c.z-.42f+t*.84f);
            }
            s.wireRenderer=MakeLine("SteelGuide",wire.transform,s.wirePoints,s.wireRadius*2,new Color(.10f,.20f,.24f));
            for(int k=0;k<2;k++)
            {
                Vector3 p=s.wirePoints[k==0?0:count-1];float h=p.y-.96f;
                Cube(k==0?"StartSupport":"FinishSupport",wire.transform,new Vector3(p.x,.96f+h*.5f,p.z),new Vector3(.027f,h,.027f),dark);
                Cube(k==0?"StartMarker":"FinishMarker",wire.transform,new Vector3(p.x,.96f,p.z),new Vector3(.17f,.025f,.17f),k==0?new Color(.15f,.72f,.5f):cyan);
            }
            var ring=new GameObject("SwivelRingTool");ring.transform.SetParent(s.transform,false);s.ringRoot=ring.transform;
            Vector3[] circle=new Vector3[65];for(int k=0;k<circle.Length;k++){float a=k/(float)(circle.Length-1)*Mathf.PI*2;circle[k]=new Vector3(Mathf.Cos(a),Mathf.Sin(a),0)*s.ringRadius;}
            s.ringRenderer=MakeLine("RingRim",ring.transform,circle,s.ringTubeRadius*2,new Color(1,.7f,.15f));
            s.ringRenderer.numCornerVertices=3;
            wire.SetActive(false);ring.SetActive(false);
        }
        static GameObject Cube(string name,Transform parent,Vector3 position,Vector3 scale,Color color)
        {
            var go=GameObject.CreatePrimitive(PrimitiveType.Cube);go.name=name;go.transform.SetParent(parent,false);go.transform.localPosition=position;go.transform.localScale=scale;SetMaterial(go,color);return go;
        }
        static void SetMaterial(GameObject go,Color color)
        {
            string key=ColorUtility.ToHtmlStringRGB(color);string path=Generated+"/RobotMat_"+key+".mat";
            var mat=AssetDatabase.LoadAssetAtPath<Material>(path);
            if(!mat){mat=new Material(Shader.Find("Standard"));mat.color=color;mat.SetFloat("_Glossiness",.35f);AssetDatabase.CreateAsset(mat,path);}
            go.GetComponent<Renderer>().sharedMaterial=mat;
        }
        static LineRenderer MakeLine(string name,Transform parent,Vector3[] points,float width,Color color)
        {
            var go=new GameObject(name);go.transform.SetParent(parent,false);var line=go.AddComponent<LineRenderer>();
            string path=Generated+"/RobotLine.mat";var mat=AssetDatabase.LoadAssetAtPath<Material>(path);
            if(!mat){mat=new Material(Shader.Find("Sprites/Default"));AssetDatabase.CreateAsset(mat,path);}
            line.sharedMaterial=mat;line.useWorldSpace=false;line.positionCount=points.Length;line.SetPositions(points);line.widthMultiplier=width;line.startColor=line.endColor=color;line.numCapVertices=3;return line;
        }
        static Bounds GetBounds(GameObject go)
        {
            var renderers=go.GetComponentsInChildren<Renderer>(true);if(renderers.Length==0)return new Bounds(go.transform.position,Vector3.one);
            Bounds bounds=renderers[0].bounds;foreach(var r in renderers)bounds.Encapsulate(r.bounds);return bounds;
        }
    }
}
